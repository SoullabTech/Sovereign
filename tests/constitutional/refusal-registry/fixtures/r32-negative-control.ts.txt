// ── R32-A sabotage: due selection made contingent on the member's absence ────
export async function findDue_VIOLATION() {
  return query(
    `SELECT r.id
       FROM member_reminders r
       JOIN members m ON m.id = r.member_id
      WHERE r.delivery_at <= now()
        AND r.cancelled_at IS NULL
        AND r.delivered_at IS NULL
        AND m.last_seen < now() - interval '7 days'`,
  );
}

// ── R32-A sabotage, camelCase form: the kind one ─────────────────────────────
export function skipIfAlreadyReturned_VIOLATION(member: Member, reminder: Reminder) {
  // Reads as consideration. Requires surveillance.
  if (member.lastActiveAt > reminder.createdAt) {
    return { skipped: 'member_already_returned' };
  }
  return null;
}

export async function suppressIfRecentlyActive_VIOLATION(memberId: string) {
  const recent = await query(
    `SELECT last_activity FROM sessions WHERE member_id = $1 ORDER BY created_at DESC LIMIT 1`,
    [memberId],
  );
  return daysSinceLastVisit(recent) < 1;
}
