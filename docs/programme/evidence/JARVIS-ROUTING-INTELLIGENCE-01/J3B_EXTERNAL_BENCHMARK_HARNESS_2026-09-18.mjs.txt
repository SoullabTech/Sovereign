#!/usr/bin/env node
import { execFileSync } from 'node:child_process';
import { writeFileSync, appendFileSync, existsSync } from 'node:fs';
import { pathToFileURL } from 'node:url';

const ROOT='/private/tmp/jarvis-routing-intelligence-01-discovery-20260918';
const MODULE=ROOT + '/scripts/builder/tinker-direct.mjs';
const ATTEMPTS='/private/tmp/jarvis-routing-j3b-attempt-ledger.jsonl';
const RESULTS='/private/tmp/jarvis-routing-j3b-external-results.jsonl';
const LOCK='/private/tmp/jarvis-routing-j3b-external-benchmark.lock';

if (existsSync(LOCK) || existsSync(ATTEMPTS) || existsSync(RESULTS)) {
  console.error('REFUSED_EXISTING_J3B_LEDGER');
  process.exit(9);
}
writeFileSync(LOCK, JSON.stringify({opened_at:new Date().toISOString(), authority:'J3-B external synthetic discrimination only'})+'\n', {mode:0o600,flag:'wx'});

const { invokeTinkerDirect } = await import(pathToFileURL(MODULE).href);
const account=process.env.USER || 'soullab';
const apiKey=execFileSync('/usr/bin/security',['find-generic-password','-a',account,'-s','soullab.tinker.api','-w'],{encoding:'utf8'}).trim();
if (!apiKey) throw new Error('PROVIDER_CREDENTIAL_MISSING');

const MODELS=[
  {family:'Inkling', model:'thinkingmachines/Inkling-Small'},
  {family:'Nemotron', model:'nvidia/NVIDIA-Nemotron-3.5-Lightning-30B-A3B-BF16'},
];

const TASKS=[
{
id:'B1_CODE_BOUNDARY',
prompt:`Return ONLY one JSON object with keys verdict, defect, invariant, minimal_fix.

- every external provider requires external_network=true;
- every metered provider additionally requires provider_spend=true.
Code:

  if (!envelope.repo_read) return false;

  return true;

Assume spec.external_network=true and spec.metered_provider=true.
Identify the exact authorization defect. verdict must be FAIL. Do not discuss style.`,

function score(text, checks) {
  const low=String(text||'').toLowerCase();
  return Object.fromEntries(checks.map(([name,alts])=>[name,alts.some(x=>low.includes(x.toLowerCase()))]));
}

let callIndex=0;
for (const spec of MODELS) {
  for (const task of TASKS) {
    callIndex += 1;
    const startedAt=new Date().toISOString();
    appendFileSync(ATTEMPTS,JSON.stringify({
      call_index:callIndex,
      started_at:startedAt,
      model_family:spec.family,
      model:spec.model,
      task:task.id,
      evidence_class:'E0_TASK_TEXT',
      files:'NO FILES',
      tools:false,
      retry_allowed:false
    })+'\n',{mode:0o600});

    const t0=process.hrtime.bigint();
    let rec;
    try {
      const result=await invokeTinkerDirect({
        model:spec.model,
        prompt:task.prompt,
        apiKey,
        maxTokens:1024
      });
      const durationMs=Number(process.hrtime.bigint()-t0)/1e6;
      const checks=score(result.text,task.checks);
      rec={
        call_index:callIndex,
        completed_at:new Date().toISOString(),
        status:'success',
        task:task.id,
        model_family:spec.family,
        model:result.model,
        provider:result.provider,
        transport:result.transport,
        duration_ms:Math.round(durationMs),
        usage:result.usage ?? null,
        stop_reason:result.stop_reason ?? null,
        cost_usd:null,
        cost_source:'not_provided_by_provider_response',
        checks,
        check_passes:Object.values(checks).filter(Boolean).length,
        check_total:Object.keys(checks).length,
        response:result.text
      };
    } catch (error) {
      const durationMs=Number(process.hrtime.bigint()-t0)/1e6;
      rec={
        call_index:callIndex,
        completed_at:new Date().toISOString(),
        status:'failed',
        task:task.id,
        model_family:spec.family,
        model:spec.model,
        duration_ms:Math.round(durationMs),
        cost_usd:null,
        cost_source:'not_provided_by_provider_response',
        error:error instanceof Error ? error.message : String(error)
      };
    }
    appendFileSync(RESULTS,JSON.stringify(rec)+'\n',{mode:0o600});
    console.log(JSON.stringify({
      call_index:rec.call_index,status:rec.status,task:rec.task,model_family:rec.model_family,
      model:rec.model,duration_ms:rec.duration_ms,usage:rec.usage??null,
      check_passes:rec.check_passes??null,check_total:rec.check_total??null,error:rec.error??null
    }));
  }
}
console.log('J3B_COMPLETE calls='+callIndex);
